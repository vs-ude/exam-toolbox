\newline
\unitlength=1cm
\begin{picture}(#1,1.5)
 \unitlength=1mm
 \thicklines
 \put(0,0){\line(1,0){#1cm}} % Grundlinie
 \put(0,8){cm} % Einheit
 \multiput(0,0)(10,0){#1}{%
  \begin{picture}(0,0)
   \thicklines
   \put(0,0){\line(0,1){6}} % Zentimeterlinie links
   \put(10,0){\line(0,1){6}} % Zentimeterlinie rechts
   \thinlines
   \put(5,0){\line(0,1){4}} % Halbzentimeterlinien
   \multiput(1,0)(5,0){2}{%
    \begin{picture}(0,0)
     \multiput(0,0)(1,0){4}{\line(0,1){2}} % Millimeterlinien
    \end{picture}%
   }%
   \stepcounter{cntcm}
   \put(10,8){\arabic{cntcm}}
  \end{picture}%
 }%
\end{picture}%
}%
