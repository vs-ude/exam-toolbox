% To set arbitrary text inside a task
% Usage: `\manualText{DE Text}{EN text}`
\newcommand{\manualText}[2]{
\ifthenelse{\equal{\sprache}{de}}{#1}{#2}
}

% To set text separately for student/solution
% Usage: `\lineloesung{Question text}{Solution text}`
\newcommand{\lineloesung}[2]{%
  \escapedStringTypeout{VSEXAM: {'Typ':'FreitextLine', 'Aufgabentext': '', 'Loesung': ''}}%
  \ifthenelse{\equal{\zeigeloesung}{yes}}%
  {%
    % yes
    {\color{red}\normalfont\itshape #2}%
  }%
  {%
    % no
    #1
  }%
}%
