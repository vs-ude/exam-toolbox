%----- Output escaping for logging
{\catcode`\ =12\relax\gdef\viktorotherspace{ }}%
\def\vcodes{%
  \catcode`\ =12\relax
}%

\def\setescape{%
  \let\ =\viktorotherspace%
}
\def\escapedStringTypeout#1{\escapeStringSec{#1}}%
\newcommand\escapeStringSec[1]{%
  \begingroup%
    \bgroup\vcodes\setescape\typeout{#1}\egroup
  \endgroup
}%
